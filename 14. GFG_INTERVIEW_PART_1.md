# SQL Queries

## 1. Fetch FIRST_NAME in uppercase with alias STUDENT_NAME

```sql
SELECT UPPER(FIRST_NAME) AS STUDENT_NAME
FROM Student;
```

---

## 2. Fetch unique values of MAJOR subjects

```sql
SELECT DISTINCT MAJOR
FROM Student;
```

Alternative:

```sql
SELECT MAJOR
FROM Student
GROUP BY MAJOR;
```

---

## 3. Print first 3 characters of FIRST_NAME

```sql
SELECT SUBSTRING(FIRST_NAME,1,3)
FROM Student;
```

---

## 4. Find position of alphabet 'a' in 'Shivansh'

```sql
SELECT INSTR(LOWER(FIRST_NAME),'a')
FROM Student
WHERE FIRST_NAME = 'Shivansh';
```

---

## 5. Fetch unique MAJOR subjects and print their length

```sql
SELECT DISTINCT MAJOR, LENGTH(MAJOR)
FROM Student;
```

---

## 6. Replace 'a' with 'A' in FIRST_NAME

```sql
SELECT REPLACE(FIRST_NAME,'a','A')
FROM Student;
```

---

## 7. Combine FIRST_NAME and LAST_NAME into COMPLETE_NAME

```sql
SELECT CONCAT(FIRST_NAME,' ',LAST_NAME) AS COMPLETE_NAME
FROM Student;
```

---

## 8. Print all students ordered by FIRST_NAME ascending and MAJOR descending

```sql
SELECT *
FROM Student
ORDER BY FIRST_NAME ASC, MAJOR DESC;
```

---

## 9. Fetch students with FIRST_NAME 'Prem' or 'Shivansh'

```sql
SELECT *
FROM Student
WHERE FIRST_NAME IN ('Prem','Shivansh');
```

---

## 10. Fetch students excluding FIRST_NAME 'Prem' and 'Shivansh'

```sql
SELECT *
FROM Student
WHERE FIRST_NAME NOT IN ('Prem','Shivansh');
```

---

## 11. Students whose FIRST_NAME ends with 'a'

```sql
SELECT *
FROM Student
WHERE FIRST_NAME LIKE '%a';
```

---

## 12. Students whose FIRST_NAME ends with 'a' and contains five alphabets before it

```sql
SELECT *
FROM Student
WHERE FIRST_NAME LIKE '_____a';
```

---

## 13. Students whose GPA lies between 9.00 and 9.99

```sql
SELECT *
FROM Student
WHERE GPA BETWEEN 9.00 AND 9.99;
```

---

## 14. Count students with MAJOR 'Computer Science'

```sql
SELECT MAJOR, COUNT(*) AS TOTAL_COUNT
FROM Student
WHERE MAJOR = 'Computer Science';
```

---

## 15. Fetch students full names with GPA between 8.5 and 9.5

```sql
SELECT CONCAT(FIRST_NAME,' ',LAST_NAME) AS FULL_NAME
FROM Student
WHERE GPA BETWEEN 8.5 AND 9.5;
```

---

## 16. Number of students in each MAJOR in descending order

```sql
SELECT MAJOR, COUNT(MAJOR)
FROM Student
GROUP BY MAJOR
ORDER BY COUNT(MAJOR) DESC;
```

---

## 17. Show only odd rows from Student table

```sql
SELECT *
FROM Student
WHERE STUDENT_ID % 2 != 0;
```

---

## 18. Top 5 records ordered by highest GPA

```sql
SELECT *
FROM Student
ORDER BY GPA DESC
LIMIT 5;
```

---

## 19. Find the 5th highest GPA

```sql
SELECT *
FROM Student
ORDER BY GPA DESC
LIMIT 4,1;
```

---

## 20. Fetch MAJOR subjects with less than 4 students

```sql
SELECT MAJOR, COUNT(MAJOR) AS MAJOR_COUNT
FROM Student
GROUP BY MAJOR
HAVING COUNT(MAJOR) < 4;
```

---

## 21. Show all MAJOR subjects with number of students

```sql
SELECT MAJOR, COUNT(MAJOR) AS ALL_MAJOR
FROM Student
GROUP BY MAJOR;
```

---

## 22. Fetch MAJOR subjects with maximum GPA in each

```sql
SELECT MAJOR, MAX(GPA) AS MAX_GPA
FROM Student
GROUP BY MAJOR;
```

---

## 23. Update GPA of students in Computer Science to 7.5

```sql
UPDATE Student
SET GPA = 7.5
WHERE MAJOR = 'Computer Science';
```

---

## 24. Find average GPA for each MAJOR

```sql
SELECT MAJOR, AVG(GPA) AS AVERAGE_GPA
FROM Student
GROUP BY MAJOR;
```

---

## 25. Show top 3 students with highest GPA

```sql
SELECT *
FROM Student
ORDER BY GPA DESC
LIMIT 3;
```

---

## 26. Count students in each MAJOR with GPA greater than 7.5

```sql
SELECT MAJOR, COUNT(STUDENT_ID) AS HIGH_GPA_COUNT
FROM Student
WHERE GPA > 7.5
GROUP BY MAJOR;
```
